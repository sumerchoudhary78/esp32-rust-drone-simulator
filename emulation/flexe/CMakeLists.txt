cmake_minimum_required(VERSION 3.20)
project(aerolab_esp32 C)
set(CMAKE_C_STANDARD 17)
if(NOT FLEXE_SOURCE)
  message(FATAL_ERROR "Use python3 scripts/setup_emulator.py")
endif()
set(FLEXE_LTO OFF CACHE BOOL "" FORCE)
set(NATIVE_ARCH OFF CACHE BOOL "" FORCE)
set(FLEXE_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${FLEXE_SOURCE}" flexe EXCLUDE_FROM_ALL)
add_executable(aerolab-esp32 bridge.c)
target_include_directories(aerolab-esp32 PRIVATE "${FLEXE_SOURCE}/src")
target_link_libraries(aerolab-esp32 PRIVATE xtensa-emu-lib)
target_compile_options(aerolab-esp32 PRIVATE -Wall -Wextra -Werror)
